# Recursive and Non-Recursive Fibonacci

n = int(input("Enter number of terms: "))

# Non-Recursive Method
a, b = 0, 1

print("\nNon-Recursive Fibonacci Series:")
for i in range(n):
    print(a, end=" ")
    a, b = b, a + b

# Recursive Method
def fib(n):
    if n <= 1:
        return n
    return fib(n-1) + fib(n-2)

print("\n\nRecursive Fibonacci Series:")
for i in range(n):
    print(fib(i), end=" ")

# Complexity Analysis
print("\n\nNon-Recursive Time Complexity: O(n)")
print("Non-Recursive Space Complexity: O(1)")
print("Recursive Time Complexity: O(2^n)")
print("Recursive Space Complexity: O(n)")
