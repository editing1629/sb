
# N-Queens Problem Using Backtracking

# Step 1: Accept input from user
n = int(input("Enter number of Queens: "))
col = int(input("Enter column for first Queen (1 to n): ")) - 1

# Step 2: Create an empty chessboard
board = [[0 for i in range(n)] for i in range(n)]

# Step 3: Check whether the Queen can be placed safely
def safe(row, col):

    # Check column
    for i in range(row):
        if board[i][col] == 1:
            return False

    # Check upper-left diagonal
    for i, j in zip(range(row-1, -1, -1),
                    range(col-1, -1, -1)):
        if board[i][j] == 1:
            return False

    # Check upper-right diagonal
    for i, j in zip(range(row-1, -1, -1),
                    range(col+1, n)):
        if board[i][j] == 1:
            return False

    return True

# Step 4: Solve the problem using recursion and backtracking
def solve(row):
    if row == n:
        return True

    for c in range(n):
        if safe(row, c):
            board[row][c] = 1

            if solve(row + 1):
                return True

            # Backtrack by removing the Queen
            board[row][c] = 0

    return False

# Step 5: Place the first Queen and check the solution
if n > 0 and 0 <= col < n:
    board[0][col] = 1

    if solve(1):
        print("N-Queens Solution:")
        for row in board:
            print(row)
    else:
        print("No solution exists")
else:
    print("Invalid input")
