# Program to Generate Binomial Coefficients Using Dynamic Programming

# Step 1: Accept input from user
n = int(input("Enter the value of n: "))
k = int(input("Enter the value of k: "))

# Step 2: Create a DP table
C = [[0 for j in range(k + 1)] for i in range(n + 1)]

# Step 3: Calculate Binomial Coefficients
for i in range(n + 1):
    for j in range(min(i, k) + 1):

        # Base cases
        if j == 0 or j == i:
            C[i][j] = 1

        # Apply the recurrence relation
        else:
            C[i][j] = C[i - 1][j - 1] + C[i - 1][j]

# Step 4: Display the Result
print("Binomial Coefficient C(", n, ",", k, ") =", C[n][k])


# Sample Output:
# Enter the value of n: 5
# Enter the value of k: 2
# Binomial Coefficient C( 5 , 2 ) = 10
